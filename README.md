# **Git Bash**..
pwd (print working directory) = displays the path to the current folder;<br>
cd %directoryName% (change directory) = change directory;<br>
1) cd ~ = go to home directory;
2) сd .. = go up a level;
3) cd . = go to current directory;<br>
ls (list directory contents) = display directory contents;<br>
ls -a = display extends list with hidden files;<br>
touch %fileName% = creates a file in the current directory;<br>
mkdir %directoryName% = make directory;<br>
cp %sourceFile% %destFile% = copy file;<br>
git mv %sourceFile%​ %destFile% = Move file or directory;<br>
cat %fileName% (concatenate and print) = output file contents;<br>
rm %fileName% = remove file;<br>
rmdir %directoryName% = remove directory;<br>
rm - rf %directoryName% = delete directory with contents;<br>
%command1% && %command2% = execute multiple commands;<br>


# **GitHub**
git init = create an empty Git repository or reinitialize an existing one;<br>
git status = show the working tree status;<br>
git add %fileName% = prepare file to saving;<br>
git commit -m 'commitName' = record changes to the repository;<br>
git log = show commit logs;<br>
git remote add %repName% %URL% = link remote repository to local;<br>
git push = update remote refs along with associated objects;<br>
git clone %URL% = clone a repository into a new directory;<br>
